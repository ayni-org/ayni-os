cmake_minimum_required(VERSION 3.22)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT AYNI_SOURCE_DIR)
    message(FATAL_ERROR "Set AYNI_SOURCE_DIR to the imported Ayni SDK source directory")
endif()
get_filename_component(AYNI_SOURCE_DIR "${AYNI_SOURCE_DIR}" REALPATH)
set(EXTRA_COMPONENT_DIRS "${AYNI_SOURCE_DIR}/hosts/esp_idf")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(ayni_portable_s3)
