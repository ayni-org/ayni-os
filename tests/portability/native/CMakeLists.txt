cmake_minimum_required(VERSION 3.22)
project(ayni_portable_native LANGUAGES C CXX)
if(NOT AYNI_SOURCE_DIR)
    message(FATAL_ERROR "AYNI_SOURCE_DIR must name the imported source SDK")
endif()
include("${AYNI_SOURCE_DIR}/cmake/AyniSDK.cmake")
ayni_add_sdk(COMPONENTS portable_profile native_host)
if(TARGET CURL::libcurl OR TARGET SQLite3::SQLite3 OR TARGET ayni_host_sim)
    message(FATAL_ERROR "Independent profile imported an unrelated deployment")
endif()
add_executable(ayni-portable main.cpp ../common/suite.cpp)
target_include_directories(ayni-portable PRIVATE ../common)
target_link_libraries(ayni-portable PRIVATE Ayni::PortableProfile Ayni::NativeHost)

if(NOT AYNI_SOURCE_REVISION)
    set(AYNI_SOURCE_REVISION "unversioned")
endif()
if(NOT AYNI_SOURCE_REVISION MATCHES "^[a-zA-Z0-9_.-]+$")
    message(FATAL_ERROR "AYNI_SOURCE_REVISION must be an opaque revision label")
endif()
target_compile_definitions(ayni-portable PRIVATE AYNI_SOURCE_REVISION="${AYNI_SOURCE_REVISION}")
target_compile_options(ayni-portable PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
target_compile_features(ayni-portable PRIVATE cxx_std_17)
