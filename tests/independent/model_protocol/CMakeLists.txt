cmake_minimum_required(VERSION 3.22)
project(ayni_model_protocol_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(AYNI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${AYNI_ROOT}/cmake/AyniSDK.cmake")
ayni_add_sdk(COMPONENTS model_protocol)
# Compile the Native transport without selecting the Linux-only Native Host.
add_subdirectory("${AYNI_ROOT}/providers/openai_compat" native-provider)
include("${AYNI_ROOT}/cmake/AyniGoogleTest.cmake")
enable_testing()
add_executable(ayni_model_protocol_checks ../../unit/model_protocol_test.cpp)
target_link_libraries(ayni_model_protocol_checks PRIVATE Ayni::ModelProtocol yyjson GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(ayni_model_protocol_checks
    DISCOVERY_TIMEOUT ${AYNI_GTEST_DISCOVERY_TIMEOUT_SECONDS})
