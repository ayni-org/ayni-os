cmake_minimum_required(VERSION 3.22)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT AYNI_SOURCE_DIR)
    get_filename_component(AYNI_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." REALPATH)
endif()
set(EXTRA_COMPONENT_DIRS "${AYNI_SOURCE_DIR}/hosts/esp_idf")
option(AYNI_WIFI_PROFILE "Build the independent touchscreen Wi-Fi profile" ON)
if(NOT AYNI_WIFI_PROFILE)
    message(FATAL_ERROR "The active experiment is Wi-Fi voice; the USB conversation profile was retired")
endif()
if(AYNI_WIFI_PROFILE)
    list(APPEND EXTRA_COMPONENT_DIRS "$ENV{IDF_PATH}/examples/bluetooth/nimble/common/nimble_central_utils")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(ayni_wifi_chat)
